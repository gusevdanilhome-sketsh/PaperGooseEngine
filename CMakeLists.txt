#=============================================================================
# CMakeLists.txt для PaperGoose Engine
#=============================================================================
# Этот файл конфигурирует сборку движка PaperGoose:
#   - проверяет компилятор (только GCC)
#   - устанавливает стандарт C17
#   - подключает GLFW через FetchContent
#   - опционально подключает Vulkan
#   - настраивает флаги компиляции отдельно для нашего кода и для внешних библиотек
#   - добавляет тесты (если включены)
#=============================================================================

#-----------------------------------------------------------------------------
# Минимальная версия CMake (3.21 требуется для поддержки C17)
#-----------------------------------------------------------------------------
cmake_minimum_required(VERSION 3.21)

#-----------------------------------------------------------------------------
# Описание проекта
#-----------------------------------------------------------------------------
project(PaperGoose
    VERSION 0.1.0
    DESCRIPTION "2D game engine with pixel-art editor, destructible physics, and ECS"
    LANGUAGES C               # Используем только язык C (C++ не нужен)
)

#-----------------------------------------------------------------------------
# Стандарт языка C17 (строгий режим, без расширений компилятора)
#-----------------------------------------------------------------------------
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)   # обязательно использовать C17
set(CMAKE_C_EXTENSIONS OFF)         # запрещаем GNU-расширения

#-----------------------------------------------------------------------------
# Проверка компилятора – необходим GCC (для совместимости с Vulkan и GLFW)
#-----------------------------------------------------------------------------
if(NOT CMAKE_C_COMPILER_ID STREQUAL "GNU")
    message(FATAL_ERROR "PaperGoose требует компилятор GCC. Обнаружен: ${CMAKE_C_COMPILER_ID}")
endif()

#-----------------------------------------------------------------------------
# Пользовательские опции (можно передать из командной строки: -DUSE_VULKAN=OFF)
#-----------------------------------------------------------------------------
option(USE_VULKAN   "Enable Vulkan renderer" ON)   # Включить Vulkan (по умолч. ВКЛ)
option(BUILD_TESTS  "Build unit tests"       ON)   # Включить тесты

#=============================================================================
# НАСТРОЙКИ ДЛЯ РАЗНЫХ ПЛАТФОРМ И ТИПОВ СБОРКИ (Debug / Release)
#=============================================================================

#-----------------------------------------------------------------------------
# Базовые флаги компиляции (зависят от типа сборки)
#-----------------------------------------------------------------------------
#   Debug   : добавляем отладочную информацию (-g)
#   Release : максимальная оптимизация по скорости (-O3)
set(CMAKE_C_FLAGS_DEBUG   "-g")
set(CMAKE_C_FLAGS_RELEASE "-O3")

#-----------------------------------------------------------------------------
# Windows (MinGW): статическая линковка библиотек GCC
#   -static-libgcc, -static-libstdc++ : встраиваем библиотеки GCC в исполняемый файл
#   -static : полная статическая линковка всех зависимостей
#-----------------------------------------------------------------------------
if(MINGW)
    set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -static-libgcc -static-libstdc++")
    set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -static")
endif()

#-----------------------------------------------------------------------------
# Linux: здесь можно добавить специфичные флаги (например, для санитайзеров)
# В текущей версии ничего дополнительного не требуется.
#-----------------------------------------------------------------------------
if(UNIX AND NOT APPLE)
    # (опционально) добавить флаги для Linux, если понадобится
    # Например: add_compile_definitions(_GNU_SOURCE)
endif()

#=============================================================================
# ПОИСК VULKAN SDK
#=============================================================================
# Vulkan ищется, но не является обязательным (если не найден – используем софт)
#-----------------------------------------------------------------------------
if(USE_VULKAN)
    find_package(Vulkan QUIET)      # QUIET – не выводим ошибку, если не найден
    if(Vulkan_FOUND)
        message(STATUS "Vulkan найден: ${Vulkan_VERSION}")
        message(STATUS "  include: ${Vulkan_INCLUDE_DIRS}")
        message(STATUS "  library: ${Vulkan_LIBRARIES}")
        add_compile_definitions(USE_VULKAN=1)   # макрос для кода (Vulkan включён)
    else()
        message(WARNING "Vulkan не найден! Будет использован софтовый рендерер.")
        add_compile_definitions(USE_VULKAN=0)   # макрос: Vulkan отключён
    endif()
else()
    message(STATUS "Vulkan рендерер отключён пользователем.")
    add_compile_definitions(USE_VULKAN=0)
endif()

#=============================================================================
# ПОДКЛЮЧЕНИЕ GLFW (через FetchContent – автоматическая загрузка и сборка)
#=============================================================================
# GLFW будет скачан из Git-репозитория и собран как статическая библиотека.
#-----------------------------------------------------------------------------
include(FetchContent)

FetchContent_Declare(
    glfw
    GIT_REPOSITORY https://github.com/glfw/glfw.git
    GIT_TAG 3.4                         # стабильный релиз GLFW 3.4
)

# Настройки сборки GLFW (отключаем всё лишнее)
set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "")   # не собирать примеры
set(GLFW_BUILD_TESTS    OFF CACHE BOOL "")   # не собирать тесты GLFW
set(GLFW_BUILD_DOCS     OFF CACHE BOOL "")   # не генерировать документацию
set(GLFW_INSTALL        OFF CACHE BOOL "")   # не устанавливать GLFW в систему

FetchContent_MakeAvailable(glfw)   # скачать, сконфигурировать и добавить цель glfw

#=============================================================================
# ИСХОДНЫЕ ФАЙЛЫ ДВИЖКА
#=============================================================================
set(SOURCES
    src/main.c
)

#=============================================================================
# СОЗДАНИЕ ИСПОЛНЯЕМОГО ФАЙЛА
#=============================================================================
add_executable(papergoose ${SOURCES})

#=============================================================================
# СТРОГИЕ ФЛАГИ КОМПИЛЯЦИИ – ПРИМЕНЯЕМ ТОЛЬКО К НАШЕМУ КОДУ (НЕ К GLFW)
#=============================================================================
#   -Wall -Wextra -Wpedantic : все основные предупреждения
#   -Werror                   : превращать предупреждения в ошибки (чистый код)
#-----------------------------------------------------------------------------
target_compile_options(papergoose PRIVATE 
    -Wall -Wextra -Wpedantic -Werror
)

#=============================================================================
# ЛИНКОВКА С БИБЛИОТЕКАМИ
#=============================================================================

# GLFW (обязательная библиотека для создания окна и ввода)
target_link_libraries(papergoose glfw)

#-----------------------------------------------------------------------------
# Windows: GLFW требует OpenGL и GDI
#-----------------------------------------------------------------------------
if(WIN32)
    target_link_libraries(papergoose opengl32 gdi32)
endif()

#-----------------------------------------------------------------------------
# Linux: GLFW обычно линкуется с X11, но CMake сам добавляет нужные библиотеки
#       (дополнительных действий не требуется)
#-----------------------------------------------------------------------------

#-----------------------------------------------------------------------------
# Vulkan (если найден и включён)
#-----------------------------------------------------------------------------
if(USE_VULKAN AND Vulkan_FOUND)
    target_link_libraries(papergoose ${Vulkan_LIBRARIES})
    target_include_directories(papergoose PRIVATE ${Vulkan_INCLUDE_DIRS})
endif()

#=============================================================================
# ТЕСТЫ (опционально)
#=============================================================================
if(BUILD_TESTS)
    enable_testing()               # включаем поддержку тестов CTest
    add_subdirectory(tests)        # подключаем tests/CMakeLists.txt
endif()

#=============================================================================
# ВЫВОД ИНФОРМАЦИИ О КОНФИГУРАЦИИ
#=============================================================================
message(STATUS "PaperGoose ${PROJECT_VERSION} настроена:")
message(STATUS "    Компилятор: ${CMAKE_C_COMPILER_ID} ${CMAKE_C_COMPILER_VERSION}")
message(STATUS "    Стандарт C: C${CMAKE_C_STANDARD}")
message(STATUS "    Vulkan: ${USE_VULKAN}")
if(USE_VULKAN AND Vulkan_FOUND)
    message(STATUS "    Vulkan версия: ${Vulkan_VERSION}")
endif()
message(STATUS "    Тесты: ${BUILD_TESTS}")
message(STATUS "    Тип сборки: ${CMAKE_BUILD_TYPE}")   # показывает Debug или Release
